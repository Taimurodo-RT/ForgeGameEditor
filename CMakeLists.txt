cmake_minimum_required(VERSION 3.24)
project(Forge VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Sources and string literals are UTF-8 (Russian text in tests, data, UI).
if(MSVC)
  add_compile_options(/utf-8)
endif()

option(FORGE_PROFILE "Build with the Tracy profiler client" ON)
option(FORGE_BUILD_TESTS "Build unit tests" ON)
option(FORGE_BUILD_BENCH "Build benchmarks" ON)

include(cmake/Dependencies.cmake)
include(cmake/Warnings.cmake)
include(cmake/Shaders.cmake)
include(cmake/Embed.cmake)

add_subdirectory(tools/shaderc)

add_subdirectory(engine/core)
add_subdirectory(engine/data)
add_subdirectory(engine/assets)
add_subdirectory(engine/world)
add_subdirectory(engine/scene)
add_subdirectory(engine/sim)
add_subdirectory(engine/script)
add_subdirectory(engine/platform)
add_subdirectory(engine/render)
add_subdirectory(engine/ui)
add_subdirectory(engine/editor)
add_subdirectory(engine/audio)
add_subdirectory(engine/game)
add_subdirectory(engine/objects)
add_subdirectory(engine/logic)
add_subdirectory(engine/level)
add_subdirectory(engine/modules)
add_subdirectory(apps/common)
add_subdirectory(apps/sandbox)
add_subdirectory(apps/world_demo)
add_subdirectory(apps/render_demo)
add_subdirectory(apps/sim_demo)
add_subdirectory(apps/factory_demo)
add_subdirectory(apps/script_demo)
add_subdirectory(apps/ui_demo)
add_subdirectory(apps/game)
add_subdirectory(apps/builtin)
add_subdirectory(apps/probe)
add_subdirectory(apps/editor)
add_subdirectory(apps/slice)

if(FORGE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(FORGE_BUILD_BENCH)
  add_subdirectory(bench)
endif()
